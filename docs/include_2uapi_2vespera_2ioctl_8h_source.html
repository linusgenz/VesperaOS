22#ifndef VESPERAOS_IOCTL_H
23#define VESPERAOS_IOCTL_H
24
25#define IOC_DIR_BITS 2
26#define IOC_SIZE_BITS 14
27#define IOC_TYPE_BITS 8
28#define IOC_NR_BITS 8
29
30#define IOC_DIR_SHIFT 30
31#define IOC_SIZE_SHIFT 16
32#define IOC_TYPE_SHIFT 8
33#define IOC_NR_SHIFT 0
34
35/* Direction flags */
36#define IOC_NONE 0
37#define IOC_WRITE 1
38#define IOC_READ 2
39
40/* Bit masks */
41#define IOC_DIR_MASK ((1u << IOC_DIR_BITS) - 1)
42#define IOC_SIZE_MASK ((1u << IOC_SIZE_BITS) - 1)
43#define IOC_TYPE_MASK ((1u << IOC_TYPE_BITS) - 1)
44#define IOC_NR_MASK ((1u << IOC_NR_BITS) - 1)
45
46#define IOC(dir, type, nr, size) \
47(((dir & IOC_DIR_MASK) << IOC_DIR_SHIFT) | \
48((size & IOC_SIZE_MASK) << IOC_SIZE_SHIFT) | \
49((type & IOC_TYPE_MASK) << IOC_TYPE_SHIFT) | \
50((nr & IOC_NR_MASK) << IOC_NR_SHIFT))
51
52#define IO(type, nr) \
53IOC(IOC_NONE, (type), (nr), 0)
54
55#define IOR(type, nr, data_type) \
56IOC(IOC_READ, (type), (nr), sizeof(data_type))
57
58#define IOW(type, nr, data_type) \
59IOC(IOC_WRITE, (type), (nr), sizeof(data_type))
60
61#define IOWR(type, nr, data_type) \
62IOC(IOC_READ | IOC_WRITE, (type), (nr), sizeof(data_type))
63
64
65#define _IO(type, nr) IO(type, nr)
66#define _IOR(type, nr, data_type) IOR(type, nr, data_type)
67#define _IOW(type, nr, data_type) IOW(type, nr, data_type)
68#define _IOWR(type, nr, data_type) IOWR(type, nr, data_type)
69
70
71/* Decoder */
72#define IOC_GET_DIR(cmd) ((uint32_t)((cmd >> IOC_DIR_SHIFT) & IOC_DIR_MASK))
73#define IOC_GET_SIZE(cmd) ((uint32_t)((cmd >> IOC_SIZE_SHIFT) & IOC_SIZE_MASK))
74#define IOC_GET_TYPE(cmd) ((uint32_t)((cmd >> IOC_TYPE_SHIFT) & IOC_TYPE_MASK))
75#define IOC_GET_NR(cmd) ((uint32_t)((cmd >> IOC_NR_SHIFT) & IOC_NR_MASK))
76
77#endif // VESPERAOS_IOCTL_H
