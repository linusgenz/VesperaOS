58/* =========================================================================
59 * Shared Memory (SHM) Low-Level Compositor Protocol
60 * ========================================================================= */
61
62#define CREP_SYNC_SHM_NAME "/crep_sync"
63#define CREP_FB_SHM_NAME "/crep_fb"
64#define CREP_MAGIC 0x43524550UL /* "CREP" */
65#define CREP_BPP 4 /* ARGB8888 */
66
67typedef struct crep_sync {
68 /* Written by server, read by client */
69 volatile uint32_t magic; /* CREP_MAGIC when server is ready */
70 volatile uint32_t ready; /* 1 after SHM initialization */
71
72 /* Geometry constants (valid after ready == 1) */
73 uint32_t width;
74 uint32_t height;
75 uint32_t bpp;
76 uint32_t pitch; /* Bytes per row (width * bpp) */
77
78 /* Written by client, read by server */
79 volatile uint32_t dirty; /* Set to 1 when a new frame is ready */
80 volatile uint32_t seq; /* Incremented on every completed frame */
81
82 uint32_t _pad[8]; /* Reserved structural alignment padding */
83} crep_sync_t;
84
85static inline uint32_t crep_fb_size(const crep_sync_t* s) {
86 return s->pitch * s->height;
87}
