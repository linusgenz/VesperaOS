31#define CREP_MAGIC 0x43524550UL // "CREP"
32#define CREP_BPP 4 // bytes per pixel, ARGB8888
33
34typedef struct crep_sync {
35 // written by server, read by client
36 volatile uint32_t magic; // CREP_MAGIC once server is ready
37 volatile uint32_t ready; // 1 after server has initialized the FB SHM
38
39 // geometry – constant after ready = 1
40 uint32_t width; // framebuffer width in pixels
41 uint32_t height; // framebuffer height in pixels
42 uint32_t bpp; // bytes per pixel (= CREP_BPP = 4)
43 uint32_t pitch; // bytes per row = width * bpp
44
45 // written by client, read by server
46 volatile uint32_t dirty; // client sets to 1 when a new frame is ready
47 volatile uint32_t seq; // client increments each time it finishes a frame
48
49 uint32_t _pad[8]; // reserved, keep struct size at 64 bytes
50} crep_sync_t;
51
52// Size of the pixel buffer SHM given a sync structure.
53static inline uint32_t crep_fb_size(const crep_sync_t* s) {
54 return s->pitch * s->height;
55}
