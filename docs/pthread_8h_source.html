49/* Futex-backed mutex.
50 * 0 = unlocked
51 * 1 = locked, no waiters
52 * 2 = locked, waiters may be sleeping on futex_wait
53 * This lets the fast (uncontended) path do a single atomic CAS with no
54 * syscall, and only pay for futex_wake when a waiter might actually exist. */
55typedef struct pthread_mutex_t {
56 uint32_t state;
57 int type; /* PTHREAD_MUTEX_* */
58 UnitID owner; /* 0 = unowned; used only for PTHREAD_MUTEX_ERRORCHECK/RECURSIVE */
59 uint32_t rec_count; /* recursion depth for PTHREAD_MUTEX_RECURSIVE */
60} pthread_mutex_t;
61
62typedef struct pthread_mutexattr_t {
63 int type;
64} pthread_mutexattr_t;
65
66typedef struct pthread_cond_t {
67 uint32_t seq;
68 clockid_t clock; /* clock used for pthread_cond_timedwait's abstime; set at
69 * init time from the condattr (or CLOCK_REALTIME by default,
70 * per POSIX). Only CLOCK_REALTIME and CLOCK_MONOTONIC are
71 * meaningful here. */
72} pthread_cond_t;
73
74typedef struct pthread_condattr_t {
75 clockid_t clock;
76} pthread_condattr_t;
77
78typedef struct pthread_barrier_t {
79 uint32_t count; /* threads required to trip the barrier */
80 uint32_t waiting; /* threads currently blocked */
81 uint32_t generation; /* bumped each time the barrier trips, for futex_wait correctness */
82 pthread_mutex_t lock;
83} pthread_barrier_t;

85typedef struct pthread_barrierattr_t {

87} pthread_barrierattr_t;

89typedef struct pthread_once_t {
