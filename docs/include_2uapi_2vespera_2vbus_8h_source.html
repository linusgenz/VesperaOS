31// Message types
32#define VBUS_MSG_SIGNAL 0x01 // One-way broadcast, no reply
33#define VBUS_MSG_CALL 0x02 // Method call (reply expected)
34#define VBUS_MSG_RETURN 0x03 // Reply to a CALL
35#define VBUS_MSG_ERROR 0x04

37#define VBUS_IFACE_POWER "vespera.power"

42#define VBUS_SIG_BATTERY_CHANGED "BatteryChanged" // payload: vbus_battery_t

48// Subscribe flags
49#define VBUS_SUB_WILDCARD 0x01 // empty member = subscribe all members of interface

51typedef struct vbus_header {
52 u32 magic; // VBUS_MAGIC
53 u8 type; // VBUS_MSG_*
54 u8 flags;
55 uint16_t header_size; // sizeof(vbus_header_t) for forward compat
56 u32 payload_size; // bytes that follow this header
57 u64 serial; // monotonic counter, set by emitter
58 u64 reply_serial; // serial of the CALL being answered; 0 for signals
59 char interface[48]; // e.g. "vespera.power"
60 char member[48]; // e.g. "BatteryChanged"
61 char sender[32]; // "kernel" or realm name
62} vbus_header_t;

65typedef struct vbus_battery {
66 u8 percent; // 0–100; 255 = unknown (battery removed)
67 u8 charging; // 1 = charging, 0 = discharging
68 u8 present; // 1 = battery present
69 u8 critical; // 1 = critically low (≤5%)
70 u32 remaining_mwh; // remaining energy in mWh; 0xFFFFFFFF = unknown
71 u32 rate_mw; // current charge/discharge rate in mW; 0 = unknown
72 u32 full_capacity_mwh; // last full capacity for reference
73 u8 index; // battery index (bat0=0, bat1=1, ...)
74 u8 _pad[3];
75} vbus_battery_t;
