82static inline int vbus_recv(vbus_header_t* hdr, void* payload_buf, size_t payload_cap) {
83 if (!hdr) return -22; // EINVAL
84
85 // Read header
86 int r = (int)sys_channel_recv(HANDLE_VBUS, (uint64_t)hdr, sizeof(vbus_header_t), 0, 0, 0);
87 if (r < 0) return r; // -EAGAIN if empty
88 if ((uint32_t)r < sizeof(vbus_header_t)) return -5; // EIO, truncated
89
90 // Validate magic
91 if (hdr->magic != VBUS_MAGIC) return -5;
92
93 // Read payload
94 if (hdr->payload_size > 0) {
95 if (payload_buf && payload_cap >= hdr->payload_size) {
96 sys_channel_recv(HANDLE_VBUS, (uint64_t)payload_buf, hdr->payload_size, 0, 0, 0);
97 } else {
98 // Drain payload into a small scratch buffer to keep the channel in sync
99 uint8_t scratch[64];
100 uint32_t remaining = hdr->payload_size;
101 while (remaining > 0) {
102 uint32_t chunk = remaining < sizeof(scratch) ? remaining : sizeof(scratch);
103 int dr = (int)sys_channel_recv(HANDLE_VBUS, (uint64_t)scratch, chunk, 0, 0, 0);
104 if (dr <= 0) break;
105 remaining -= (uint32_t)dr;
106 }
107 }
108 }
109 return 1;
110}
