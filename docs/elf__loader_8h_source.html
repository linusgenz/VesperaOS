23// ---------------------------------------------------------------------------
24// Freistehender ELF64-Loader-Kern: Mapping, DT_*-Scan, Relokation,
25// Symbolsuche. Wird sowohl von dlfcn.c (Runtime dlopen()) als auch von
26// ld-vespera.so (Bootstrap-Interpreter fuer PT_INTERP-Programme) genutzt.
27//
28// Diese Datei setzt voraus, dass folgende Symbole beim Linken verfuegbar
29// sind (siehe Forward-Declarations unten):
30// - memcpy, memset (freistehende oder vesplib-Variante)
31// - elf_loader_sys_mmap, elf_loader_sys_mprotect, elf_loader_sys_munmap
32// Letztere drei sind bewusst NICHT direkt mmap/mprotect/munmap, sondern ein
33// duenner Name, den der Aufrufer (dlfcn.c bzw. ld_main.c) auf den jeweils
34// passenden Syscall-Wrapper mapped - so bleibt diese Datei unabhaengig
35// davon, ob gerade vesplib-Wrapper oder rohe Syscalls verfuegbar sind.
36// ---------------------------------------------------------------------------
37

78// ---------------------------------------------------------------------------
79// Ein einzelnes geladenes ELF-Objekt (Hauptprogramm oder Shared Object).
80// Enthaelt alles, was fuer Relokation und Symbolsuche gebraucht wird.
81// dlfcn.c bettet dies in sein dl_object_t ein bzw. nutzt es 1:1; ld_main.c
82// haelt ein flaches Array davon fuer die DT_NEEDED-Kette.
83// ---------------------------------------------------------------------------
84typedef struct elf_loaded_object {
85 uint64_t load_bias;
86 uint64_t map_base;
87 uint64_t map_size;
88 Elf64_Ehdr* ehdr;
89
90 Elf64_Sym* dynsym;
91 const char* dynstr;
92 uint64_t dynsym_count;
93
94 uint32_t* hash; // DT_HASH, falls vorhanden (sonst NULL)
95 uint32_t* gnu_hash; // DT_GNU_HASH, falls vorhanden (sonst NULL)
96
97 void* init_fn; // DT_INIT
98 void** init_array; // DT_INIT_ARRAY
99 uint64_t init_array_count;
100
101 // Aufloesungskette fuer Symbolsuche (DT_NEEDED-Abhaengigkeiten). Der
102 // Aufrufer befuellt dies (Speicherverwaltung bleibt bei ihm - dlfcn.c
103 // nutzt sein festes Slot-Array, ld_main.c ein eigenes flaches Array).
104 struct elf_loaded_object* needed[ELF_LOADER_MAX_NEEDED];
105 uint32_t needed_count;
106} elf_loaded_object_t;

108// Ergebnis von elf_loader_scan_dynamic(): alles aus PT_DYNAMIC, was fuer
109// Relokation und DT_NEEDED-Aufloesung noetig ist, aber noch keine
110// Objekt-Referenzen braucht (die DT_NEEDED-Eintraege sind zunaechst nur
111// String-Offsets - Aufloesung/Laden der jeweiligen Datei bleibt Sache des
112// Aufrufers, da das File-I/O erfordert).
113typedef struct {
114 uint64_t rela_addr, rela_size, rela_ent;
115 uint64_t jmprel_addr, jmprel_size;
116 uint64_t needed_off[ELF_LOADER_MAX_NEEDED];
117 uint32_t needed_count;
118} elf_dyn_scan_result_t;

139// Findet das PT_DYNAMIC-Programmheader-Segment. NULL, falls nicht vorhanden

143// Findet das PT_INTERP-Programmheader-Segment. NULL, falls nicht vorhanden.

147// --- Mapping -------------------------------------------------------------
148
149// Mapped alle PT_LOAD-Segmente von `file` relativ zu load_bias in den
150// zuvor reservierten Bereich [map_base, map_base+map_size). Kopiert
151// Dateiinhalt hinein und setzt anschliessend per elf_loader_sys_mprotect
152// die tatsaechlichen Segment-Schutzrechte (R/W/X gemaess p_flags). Gibt 1
153// bei Erfolg zurueck, 0 wenn das initiale Reservierungs-Mapping fehlschlaegt.
154int elf_loader_map_segments(const Elf64_Ehdr* eh, const void* file,
155 uint64_t load_bias, uint64_t map_base,
156 uint64_t map_size);
157
158// --- PT_DYNAMIC-Scan -------------------------------------------------------
159
160// Durchlaeuft PT_DYNAMIC und befuellt sowohl obj (dynsym/dynstr/hash/
161// init_fn/init_array/...) als auch out (rela/jmprel-Lage, DT_NEEDED-Offsets).
162// dyn zeigt bereits auf die relozierte (load_bias-addierte) Adresse des
163// PT_DYNAMIC-Segments.
164void elf_loader_scan_dynamic(elf_loaded_object_t* obj, const Elf64_Dyn* dyn,
165 uint64_t load_bias, elf_dyn_scan_result_t* out);
166
167// --- Symbolsuche -------------------------------------------------------------
168
169// Sucht `name` ausschliesslich innerhalb von obj (kein Abstieg in
170// obj->needed). NULL, falls nicht gefunden oder kein dynsym/dynstr
171// vorhanden.
172Elf64_Sym* elf_loader_find_symbol_in_object(elf_loaded_object_t* obj, const char* name);
173
174// Sucht `name` in obj und rekursiv in obj->needed[]. *found_in wird auf das
175// Objekt gesetzt, in dem das Symbol tatsaechlich definiert ist (fuer die
176// load_bias-Berechnung der Aufrufer). depth ist der Rekursionsstart (0 beim
177// ersten Aufruf) und wird intern gegen eine feste Tiefe (16) begrenzt, um
178// zyklische DT_NEEDED-Graphen abzufangen.
179Elf64_Sym* elf_loader_find_symbol_recursive(elf_loaded_object_t* obj, const char* name,
180 elf_loaded_object_t** found_in, int depth);
181
182// --- Relokation -------------------------------------------------------------
183
184// Wendet eine RELA-Tabelle (.rela.dyn oder .rela.plt) auf obj an. Behandelt
185// R_X86_64_RELATIVE, R_X86_64_GLOB_DAT, R_X86_64_JUMP_SLOT, R_X86_64_64.
186// Symbolaufloesung fuer undefinierte Symbole laeuft ueber
187// elf_loader_find_symbol_recursive() innerhalb von obj.
188void elf_loader_apply_relocations(elf_loaded_object_t* obj, uint64_t rela_addr,
189 uint64_t rela_size, uint64_t rela_ent);
190
191// --- Init-Funktionen -------------------------------------------------------
192
193// Ruft DT_INIT und anschliessend alle Eintraege von DT_INIT_ARRAY auf.
194void elf_loader_run_init_functions(elf_loaded_object_t* obj);
