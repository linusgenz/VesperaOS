180/* How to extract and insert information held in the r_info field. */
181
182#define ELF32_R_SYM(val) ((val) >> 8)
183#define ELF32_R_TYPE(val) ((val) & 0xff)
184#define ELF32_R_INFO(sym, type) (((sym) << 8) + ((type) & 0xff))
185
186#define ELF64_R_SYM(i) ((i) >> 32)
187#define ELF64_R_TYPE(i) ((i) & 0xffffffff)
188#define ELF64_R_INFO(sym, type) ((((Elf64_Xword)(sym)) << 32) + (type))
