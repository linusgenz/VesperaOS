193 vector is not usually defined in a standard <elf.h> file, but it

278/* How to extract and insert information held in the r_info field. */
279
280#define ELF32_R_SYM(val) ((val) >> 8)
281#define ELF32_R_TYPE(val) ((val) & 0xff)
282#define ELF32_R_INFO(sym, type) (((sym) << 8) + ((type) & 0xff))
283
284#define ELF64_R_SYM(i) ((i) >> 32)
285#define ELF64_R_TYPE(i) ((i) & 0xffffffff)
286#define ELF64_R_INFO(sym, type) ((((Elf64_Xword)(sym)) << 32) + (type))
