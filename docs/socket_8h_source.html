178#define CMSG_ALIGN(len) (((len) + sizeof(size_t) - 1) & ~(sizeof(size_t) - 1))
179#define CMSG_SPACE(len) (CMSG_ALIGN(sizeof(struct cmsghdr)) + CMSG_ALIGN(len))

188static inline struct cmsghdr* cmsg_nxthdr(struct msghdr* mhdr, struct cmsghdr* cmsg) {
189 unsigned char* end = (unsigned char*)(mhdr->msg_control) + mhdr->msg_controllen;
190 unsigned char* next = (unsigned char*)cmsg + CMSG_ALIGN(cmsg->cmsg_len);
191 if (next + sizeof(struct cmsghdr) > end) return (struct cmsghdr*)0;
192 return (struct cmsghdr*)next;
193}
