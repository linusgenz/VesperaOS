69/* -------------------------------------------------------------------------
70 * Colour
71 *
72 * stella_color_t is an opaque 32-bit value. Use stella_rgb() to construct
73 * one; use the STELLA_COL() convenience macro with theme colour tokens.
74 * ------------------------------------------------------------------------- */
75
76typedef struct {
77 uint32_t _raw;
78} stella_color_t;
79
81static inline stella_color_t stella_rgb(uint8_t r, uint8_t g, uint8_t b) {
82 stella_color_t c;
83 c._raw = ((uint32_t)r << 16) | ((uint32_t)g << 8) | (uint32_t)b;
84 return c;
85}
86
88static inline stella_color_t stella_hex(uint32_t hex) {
89 stella_color_t c;
90 c._raw = hex & 0x00FFFFFFu;
91 return c;
92}
