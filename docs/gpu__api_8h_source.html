57typedef enum {
58 GPU_MEM_CACHED = 0, // CPU cached; requires explicit flush before GPU read
59 GPU_MEM_UNCACHED = 1 << 0, // Uncached access
60 GPU_MEM_WRITE_COMBINE = 1 << 1, // Optimized for CPU-write-mostly data (vertices/uniforms)
61} gpu_mem_flags_t;
