        ${CMAKE_CURRENT_SOURCE_DIR}
        ${INC_MESA}
        ${INC_INCLUDE}
        ${INC_SRC}
        ${INC_GALLIUM}
)

    target_compile_options(glapi_bridge PRIVATE ${C_MSVC_COMPAT_ARGS} ${BRIDGE_GLAPI_ARGS})
endif()

        ${DEP_THREAD}
        ${IDEP_MESAUTIL}
