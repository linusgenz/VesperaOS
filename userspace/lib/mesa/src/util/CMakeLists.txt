# ---------------------------------------------------------------------------
# format_srgb.c is generated by format_srgb.py (pure code generation; see
# util/meson.build: custom_target(‘format_srgb’, ...)). We build it
# equivalently using a CMake custom command
# ---------------------------------------------------------------------------

set(FORMAT_SRGB_C ${CMAKE_CURRENT_BINARY_DIR}/format_srgb.c)
add_custom_command(
        OUTPUT ${FORMAT_SRGB_C}
        COMMAND ${MESA_PYTHON3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/format_srgb.py > ${FORMAT_SRGB_C}
        DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/format_srgb.py
        COMMENT "Generating format_srgb.c"
        VERBATIM
)

endif()

# ---------------------------------------------------------------------------
# u_format_gen.h / u_format_pack.h / u_format_table.c are generated in Mesa upstream
# via format/meson.build from u_format_table.py + u_format.yaml
# (three custom_targets, same Python script with different flags):
#
#   u_format_gen.h    <- u_format_table.py --enums   > u_format_gen.h
#   u_format_pack.h   <- u_format_table.py --header  > u_format_pack.h
#   u_format_table.c  <- u_format_table.py           > u_format_table.c
#
# ---------------------------------------------------------------------------
set(U_FORMAT_GEN_ROOT ${CMAKE_CURRENT_BINARY_DIR}/gen)
set(U_FORMAT_GEN_DIR ${U_FORMAT_GEN_ROOT}/util/format)
file(MAKE_DIRECTORY ${U_FORMAT_GEN_DIR})

set(U_FORMAT_TABLE_PY ${CMAKE_CURRENT_SOURCE_DIR}/format/u_format_table.py)
set(U_FORMAT_YAML ${CMAKE_CURRENT_SOURCE_DIR}/format/u_format.yaml)
set(U_FORMAT_PACK_PY ${CMAKE_CURRENT_SOURCE_DIR}/format/u_format_pack.py)
set(U_FORMAT_PARSE_PY ${CMAKE_CURRENT_SOURCE_DIR}/format/u_format_parse.py)

set(U_FORMAT_GEN_H ${U_FORMAT_GEN_DIR}/u_format_gen.h)
add_custom_command(
        OUTPUT ${U_FORMAT_GEN_H}
        COMMAND ${CMAKE_COMMAND} -E env sh -c "${MESA_PYTHON3_EXECUTABLE} ${U_FORMAT_TABLE_PY} ${U_FORMAT_YAML} --enums > ${U_FORMAT_GEN_H}"
        DEPENDS ${U_FORMAT_TABLE_PY} ${U_FORMAT_YAML} ${U_FORMAT_PACK_PY} ${U_FORMAT_PARSE_PY}
        COMMENT "Generating util/format/u_format_gen.h"
        VERBATIM
)
set(U_FORMAT_GEN_H ${U_FORMAT_GEN_H} PARENT_SCOPE)

set(U_FORMAT_PACK_H ${U_FORMAT_GEN_DIR}/u_format_pack.h)
add_custom_command(
        OUTPUT ${U_FORMAT_PACK_H}
        COMMAND ${CMAKE_COMMAND} -E env sh -c "${MESA_PYTHON3_EXECUTABLE} ${U_FORMAT_TABLE_PY} ${U_FORMAT_YAML} --header > ${U_FORMAT_PACK_H}"
        DEPENDS ${U_FORMAT_TABLE_PY} ${U_FORMAT_YAML} ${U_FORMAT_PACK_PY} ${U_FORMAT_PARSE_PY}
        COMMENT "Generating util/format/u_format_pack.h"
        VERBATIM
)

set(U_FORMAT_TABLE_C ${CMAKE_CURRENT_BINARY_DIR}/u_format_table.c)
add_custom_command(
        OUTPUT ${U_FORMAT_TABLE_C}
        COMMAND ${CMAKE_COMMAND} -E env sh -c "${MESA_PYTHON3_EXECUTABLE} ${U_FORMAT_TABLE_PY} ${U_FORMAT_YAML} > ${U_FORMAT_TABLE_C}"
        DEPENDS ${U_FORMAT_TABLE_PY} ${U_FORMAT_YAML} ${U_FORMAT_PACK_PY} ${U_FORMAT_PARSE_PY}
        COMMENT "Generating u_format_table.c"
        VERBATIM
)


# u_format.c/u_format_pack.h includieren "u_format_pack.h" unqualifiziert
# (ohne util/format/-Präfix), deshalb muss U_FORMAT_GEN_DIR selbst zusätzlich
# als -I-Pfad rein, nicht nur U_FORMAT_GEN_ROOT.
list(APPEND MESA_UTIL_FORMAT_SOURCES ${U_FORMAT_TABLE_C})

# Reine Header tauchen sonst in keinem Target als Output auf, wodurch
# CMake/Ninja add_custom_command dafuer nie ausfuehrt (nur .c-Outputs, die
# tatsaechlich kompiliert werden, ziehen ihre custom_command automatisch).
# Deshalb explizit als Sources aufnehmen (werden nicht kompiliert, nur als
# Dependency getrackt).
list(APPEND MESA_UTIL_FORMAT_SOURCES ${U_FORMAT_GEN_H} ${U_FORMAT_PACK_H})

        PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}
